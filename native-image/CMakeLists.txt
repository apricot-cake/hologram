cmake_minimum_required(VERSION 3.22)
project(hologram_avif_validator LANGUAGES C)

include(FetchContent)
# 検査と再生の許可条件を固定するため、codec の選択を自動検出しない。
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_CODEC_DAV1D LOCAL CACHE STRING "" FORCE)
set(AVIF_CODEC_AOM OFF CACHE STRING "" FORCE)
set(AVIF_CODEC_LIBGAV1 OFF CACHE STRING "" FORCE)
set(AVIF_CODEC_RAV1E OFF CACHE STRING "" FORCE)
set(AVIF_CODEC_SVT OFF CACHE STRING "" FORCE)
set(AVIF_CODEC_AVM OFF CACHE STRING "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libavif
    GIT_REPOSITORY https://github.com/AOMediaCodec/libavif.git
    GIT_TAG c5240fc79fe5c2407e10afd35f5505ef6333ea49
    SOURCE_SUBDIR hologram-configure-explicitly)
# libavif v1.4.2 の LocalDav1d.cmake は dav1d 1.5.3 を取得する。
FetchContent_MakeAvailable(libavif)
# libavif は allocator 設定を公開していないため、固定版 adapter にだけ
# dav1d の公式 allocator 設定フックを追加する。取得した原本は変更しない。
set(codec_source "${libavif_SOURCE_DIR}/src/codec_dav1d.c")
file(SHA256 "${codec_source}" codec_hash)
if(NOT codec_hash STREQUAL "b29458f575e87264e11ff6f19db1b18b5bf5f19e9a603df452b22319bba15357")
    message(FATAL_ERROR "Fixed libavif dav1d adapter changed; allocator hook must be revalidated")
endif()
file(READ "${codec_source}" codec_text)
string(REPLACE "#include <string.h>" "#include <string.h>\nextern void hologramConfigureDav1dBudget(Dav1dSettings *settings);" codec_text "${codec_text}")
string(REPLACE "        dav1dSettings.logger.cookie = codec;" "        hologramConfigureDav1dBudget(&dav1dSettings);\n        dav1dSettings.logger.cookie = codec;" codec_text "${codec_text}")
set(codec_adapter "${CMAKE_CURRENT_BINARY_DIR}/hologram-codec-dav1d.c")
file(WRITE "${codec_adapter}" "${codec_text}")
find_package(Git REQUIRED)
set(dav1d_revision_check "${CMAKE_CURRENT_BINARY_DIR}/verify-dav1d-revision.cmake")
file(WRITE "${dav1d_revision_check}" "execute_process(COMMAND \"${GIT_EXECUTABLE}\" -C \"${FETCHCONTENT_BASE_DIR}/dav1d-src\" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status)\nif(NOT status EQUAL 0 OR NOT revision STREQUAL \"b546257f770768b2c88258c533da38b91a06f737\")\n  message(FATAL_ERROR \"Fixed dav1d revision mismatch\")\nendif()\n")
set(codec_module_dir "${CMAKE_CURRENT_BINARY_DIR}/cmake-adapter")
file(MAKE_DIRECTORY "${codec_module_dir}")
file(WRITE "${codec_module_dir}/LocalDav1d.cmake" "include(\"${libavif_SOURCE_DIR}/cmake/Modules/LocalDav1d.cmake\")\nif(NOT TARGET dav1d)\n  message(FATAL_ERROR \"Prebuilt dav1d is unsupported; use the fixed source build\")\nendif()\nExternalProject_Add_Step(dav1d verify_fixed_revision COMMAND \"${CMAKE_COMMAND}\" -P \"${dav1d_revision_check}\" DEPENDEES download DEPENDERS configure ALWAYS TRUE)\n")
list(PREPEND CMAKE_MODULE_PATH "${codec_module_dir}")
add_subdirectory("${libavif_SOURCE_DIR}" "${libavif_BINARY_DIR}")
get_target_property(avif_sources avif_obj SOURCES)
if(NOT "src/codec_dav1d.c" IN_LIST avif_sources)
    message(FATAL_ERROR "Fixed libavif codec source list changed")
endif()
list(REMOVE_ITEM avif_sources "src/codec_dav1d.c")
set_property(TARGET avif_obj PROPERTY SOURCES "${avif_sources};${codec_adapter}")

add_executable(hologram-avif-validator avif-validator.c)
target_compile_features(hologram-avif-validator PRIVATE c_std_11)
target_link_libraries(hologram-avif-validator PRIVATE avif)
if(WIN32)
    target_link_libraries(hologram-avif-validator PRIVATE Psapi)
endif()
target_include_directories(hologram-avif-validator PRIVATE "${FETCHCONTENT_BASE_DIR}/dav1d-install/include")
add_dependencies(hologram-avif-validator dav1d)
if(MSVC)
    target_compile_options(hologram-avif-validator PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(hologram-avif-validator PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(hologram-avif-validator PRIVATE -Wall -Wextra -Werror)
    if(WIN32)
        target_link_options(hologram-avif-validator PRIVATE -municode)
    endif()
endif()
install(TARGETS hologram-avif-validator RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(hologram-avif-allocator-test avif-validator-allocator-test.c)
    target_compile_features(hologram-avif-allocator-test PRIVATE c_std_11)
    target_link_libraries(hologram-avif-allocator-test PRIVATE avif)
    if(WIN32)
        target_link_libraries(hologram-avif-allocator-test PRIVATE Psapi)
    endif()
    target_include_directories(hologram-avif-allocator-test PRIVATE "${FETCHCONTENT_BASE_DIR}/dav1d-install/include")
    add_dependencies(hologram-avif-allocator-test dav1d)
    if(MSVC)
        target_compile_options(hologram-avif-allocator-test PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(hologram-avif-allocator-test PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(hologram-avif-allocator-test PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME avif-allocator-layout-and-budgets COMMAND hologram-avif-allocator-test)
endif()
